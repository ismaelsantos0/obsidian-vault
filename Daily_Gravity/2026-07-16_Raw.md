### Como seria na prática:

#### Passo 1: No código da sua loja (Annalicia Modas)
Se a sua loja foi feita em **Next.js** (igual ao ConectaZap), você só precisaria criar um arquivo de rota (ex: `app/api/produtos/route.ts`) com um código bem simples, mais ou menos assim:

```typescript
// Exemplo de código na sua Loja
import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma"; // Banco de dados da loja

export async function GET(request: Request) {
  // Pega a palavra que o N8N vai enviar (ex: "saia")
  const { searchParams } = new URL(request.url);
  const busca = searchParams.get("busca");

  // Procura no banco de dados interno da loja
  const produtos = await prisma.produto.findMany({
    where: {
      nome: { contains: busca, mode: "insensitive" },
      estoque: { gt: 0 } // Só produtos com estoque maior que 0
    },
    select: { nome: true, preco: true, slug: true },
    take: 5 // Limita a 5 produtos
  });

  // Devolve o resultado empacotado para o N8N
  return NextResponse.json(produtos);
}
```

```python
@router.get("/search")
async def search_produtos(busca: str = "", db: AsyncSession = Depends(get_db)):
    """Microserviço para N8N: busca produtos pelo nome que estão em estoque."""
    query = select(Produto).where(
        Produto.nome.ilike(f"%{busca}%"),
        Produto.estoque > 0,
        Produto.is_active == True
    ).limit(5)
    
    result = await db.execute(query)
    produtos = result.scalars().all()
    
    return [{"nome": p.nome, "preco": p.preco, "slug": p.slug} for p in produtos]
```

import { NextResponse } from "next/server";
import { prisma } from "@/lib/prisma"; // Banco de dados da loja
export async function GET(request: Request) {

@router.get("/search")
async def search_produtos(busca: str = "", db: AsyncSession = Depends(get_db)):
    """Microserviço para N8N: busca produtos pelo nome que estão em estoque."""
    query = select(Produto).where(
        Produto.nome.ilike(f"%{busca}%"),
        Produto.estoque > 0,
        Produto.is_active == True
    ).limit(5)
    
    result = await db.execute(query)
